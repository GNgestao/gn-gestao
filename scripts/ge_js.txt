function geRenderFerias(){
  var ano = parseInt(document.getElementById('geAnoFerias').value);
  var grid = document.getElementById('geGridFerias');

  var html = '<table style="border-collapse:collapse;width:100%;min-width:900px;font-size:.8rem;">';
  html += '<thead><tr><th style="background:#2a1a4e;color:#fff;padding:.5rem .7rem;text-align:left;position:sticky;left:0;z-index:2;min-width:130px;">Técnico</th>';
  GE_MESES.forEach(function(m){ html += '<th style="background:#2a1a4e;color:#aaa;padding:.5rem .7rem;text-align:center;min-width:90px;">'+m+'/'+String(ano).slice(2)+'</th>'; });
  html += '</tr></thead><tbody>';

  GE_TECNICOS.forEach(function(tec){
    var cor = GE_CORES[tec.grupo];
    html += '<tr>';
    html += '<td style="background:#1a0f35;color:'+cor+';font-weight:600;padding:.4rem .7rem;position:sticky;left:0;border-bottom:1px solid #2a1a4e;">'+tec.nome+'</td>';

    for(var m=0; m<12; m++){
      var feriasDoMes = _geFerias.filter(function(f){
        if(f.nome !== tec.nome) return false;
        var ini = new Date(f.inicio);
        var fim = new Date(f.fim);
        var mesIni = new Date(ano, m, 1);
        var mesFim = new Date(ano, m+1, 0);
        return ini <= mesFim && fim >= mesIni;
      });

      var cell = '<td style="background:#12082a;padding:.3rem .5rem;text-align:center;border-bottom:1px solid #1a0f35;border-left:1px solid #1a0f35;">';
      if(feriasDoMes.length > 0){
        feriasDoMes.forEach(function(f){
          var ini = new Date(f.inicio);
          var fim = new Date(f.fim);
          var dIni = String(ini.getDate()).padStart(2,'0')+'/'+String(ini.getMonth()+1).padStart(2,'0');
          var dFim = String(fim.getDate()).padStart(2,'0')+'/'+String(fim.getMonth()+1).padStart(2,'0');
          cell += '<div onclick="geAbrirModalFerias(\'' + f.id + '\')" style="cursor:pointer;display:inline-block;background:'+cor+'22;border:1px solid '+cor+';border-radius:4px;padding:.1rem .3rem;margin:.1rem;">';
          cell += '<span style="color:#22C55E;font-size:.75rem;">'+dIni+'</span>';
          cell += '<span style="color:#aaa;font-size:.7rem;"> até </span>';
          cell += '<span style="color:#E84A1A;font-size:.75rem;">'+dFim+'</span>';
          cell += '</div>';
        });
      } else {
        cell += '<span style="color:#333;">—</span>';
      }
      cell += '</td>';
      html += cell;
    }
    html += '</tr>';
  });

  html += '</tbody></table>';
  grid.innerHTML = html;
}

function geAbrirModalFerias(id){
  var modal = document.getElementById('geModalFerias');
  var sel = document.getElementById('geFeriasNome');
  sel.innerHTML = GE_TECNICOS.map(function(t){ return '<option value="'+t.nome+'">'+t.nome+'</option>'; }).join('');

  if(id){
    var f = _geFerias.find(function(x){ return x.id===id; });
    if(f){
      sel.value = f.nome;
      document.getElementById('geFeriasInicio').value = f.inicio;
      document.getElementById('geFeirasFim').value = f.fim;
      _geEditandoFerias = id;
    }
  } else {
    sel.value = GE_TECNICOS[0].nome;
    document.getElementById('geFeriasInicio').value = '';
    document.getElementById('geFeirasFim').value = '';
    _geEditandoFerias = null;
  }
  modal.style.display = 'flex';
}

function geFecharModalFerias(){
  document.getElementById('geModalFerias').style.display = 'none';
  _geEditandoFerias = null;
}

function geSalvarFerias(){
  var nome = document.getElementById('geFeriasNome').value;
  var inicio = document.getElementById('geFeriasInicio').value;
  var fim = document.getElementById('geFeirasFim').value;
  if(!inicio || !fim){ alert('Informe início e fim das férias.'); return; }
  if(inicio > fim){ alert('Data início deve ser anterior ao fim.'); return; }

  if(_geEditandoFerias){
    var idx = _geFerias.findIndex(function(x){ return x.id===_geEditandoFerias; });
    if(idx>=0) _geFerias[idx] = {id:_geEditandoFerias, nome:nome, inicio:inicio, fim:fim};
  } else {
    _geFerias.push({id:'fe_'+Date.now(), nome:nome, inicio:inicio, fim:fim});
  }
  localStorage.setItem('ge_ferias', JSON.stringify(_geFerias));
  geFecharModalFerias();
  geRenderFerias();
}

function geTecnicoEmFerias(nome, data){
  var d = new Date(data);
  return _geFerias.some(function(f){
    return f.nome===nome && new Date(f.inicio)<=d && new Date(f.fim)>=d;
  });
}

// Gerador de Escala
function geGerarEscala(){
  var ano = parseInt(document.getElementById('geAnoEscala').value);
  var semanas = [];

  // Calcular todos os fins de semana do ano
  var d = new Date(ano, 0, 1);
  while(d.getDay() !== 6) d.setDate(d.getDate()+1); // primeiro sábado

  var ciclo = {laranja:0, azul:0, verde:0}; // controle 2x1

  while(d.getFullYear() === ano){
    var sabado = new Date(d);
    var domingo = new Date(d); domingo.setDate(domingo.getDate()+1);
    var numSemana = semanas.length + 1;

    var sSab = sabado.toISOString().split('T')[0];
    var sDom = domingo.toISOString().split('T')[0];

    var semana = {
      numSemana: numSemana,
      sabado: sSab,
      domingo: sDom,
      plantoes: geAlocarPlantoes(sSab, sDom, ciclo, numSemana)
    };
    semanas.push(semana);
    d.setDate(d.getDate()+7);
  }

  geRenderEscala(semanas, ano);
}

function geGrupoDisponivel(grupo, ciclo){
  // 2x1: trabalha 2 semanas, folga 1
  var c = ciclo[grupo] % 3;
  return c < 2; // 0 e 1 trabalha, 2 folga
}

function geAlocarPlantoes(sSab, sDom, ciclo, numSemana){
  var plantoes = [];

  // Avançar ciclos dos grupos
  var trabLaranja = geGrupoDisponivel('laranja', ciclo);
  var trabAzul = geGrupoDisponivel('azul', ciclo);
  var trabVerde = geGrupoDisponivel('verde', ciclo);

  // SÁBADO
  // SRM 1° turno — Bruno (verde) se disponível
  if(trabVerde && !geTecnicoEmFerias('Bruno', sSab)){
    plantoes.push({dia:'SAB', setor:'SRM', tecnico:'Bruno', horario:'08:00-17:00'});
  }
  // SRM 2° turno — Humberto (laranja)
  if(trabLaranja && !geTecnicoEmFerias('Humberto', sSab)){
    plantoes.push({dia:'SAB', setor:'SRM', tecnico:'Humberto', horario:'13:00-22:00'});
  }
  // SBV — Paulo André (verde), subs Alisson, Giliard, Charlington
  var sbvTec = null;
  if(trabVerde && !geTecnicoEmFerias('Paulo Andre', sSab)) sbvTec='Paulo Andre';
  else if(trabAzul && !geTecnicoEmFerias('Alisson', sSab)) sbvTec='Alisson';
  else if(trabLaranja && !geTecnicoEmFerias('Giliard', sSab)) sbvTec='Giliard';
  else if(trabLaranja && !geTecnicoEmFerias('Charlington', sSab)) sbvTec='Charlington';
  if(sbvTec) plantoes.push({dia:'SAB', setor:'SBV', tecnico:sbvTec, horario:'08:00-17:00'});

  // SPO sábado — Wellington/Klebson Ramos alternando
  var spoTec = null;
  if(numSemana%2===0){
    if(trabAzul && !geTecnicoEmFerias('Wellington', sSab)) spoTec='Wellington';
    else if(trabLaranja && !geTecnicoEmFerias('Klebson Ramos', sSab)) spoTec='Klebson Ramos';
  } else {
    if(trabLaranja && !geTecnicoEmFerias('Klebson Ramos', sSab)) spoTec='Klebson Ramos';
    else if(trabAzul && !geTecnicoEmFerias('Wellington', sSab)) spoTec='Wellington';
  }
  if(spoTec) plantoes.push({dia:'SAB', setor:'SPO', tecnico:spoTec, horario:'10:00-21:00'});

  // ERM — Adriano Rog (fixo)
  if(!geTecnicoEmFerias('Adriano Rog', sSab)){
    plantoes.push({dia:'SAB', setor:'ERM', tecnico:'Adriano Rog', horario:'08:00-12:00'});
  }
  // TRF — Laercio e Marcelo (fixos)
  if(!geTecnicoEmFerias('Laercio', sSab)){
    plantoes.push({dia:'SAB', setor:'TRF', tecnico:'Laercio', horario:'08:00-12:00'});
  }
  if(!geTecnicoEmFerias('Marcelo', sSab)){
    plantoes.push({dia:'SAB', setor:'TRF', tecnico:'Marcelo', horario:'12:00-17:00'});
  }
  // RHP sábado dia — demais disponíveis
  ['Rodolfo','Durval','Elenildo','Joaz','Tone','Evaldo'].forEach(function(nome){
    var tec = GE_TECNICOS.find(function(t){ return t.nome===nome; });
    if(!tec) return;
    var trab = tec.grupo==='laranja'?trabLaranja:tec.grupo==='azul'?trabAzul:tec.grupo==='verde'?trabVerde:true;
    if(trab && !geTecnicoEmFerias(nome, sSab)){
      plantoes.push({dia:'SAB', setor:'RHP', tecnico:nome, horario:'07:00-19:00'});
    }
  });

  // DOMINGO
  // SRM dom — Klebson Andrade
  if(trabLaranja && !geTecnicoEmFerias('Klebson Andrade', sDom)){
    plantoes.push({dia:'DOM', setor:'SRM', tecnico:'Klebson Andrade', horario:'10:00-21:20'});
  }
  // RHP dom — George
  if(trabVerde && !geTecnicoEmFerias('George', sDom)){
    plantoes.push({dia:'DOM', setor:'RHP', tecnico:'George', horario:'07:00-19:00'});
  }
  // SPO dom — Adriano Francisco
  if(trabAzul && !geTecnicoEmFerias('Adriano Fran', sDom)){
    plantoes.push({dia:'DOM', setor:'SPO', tecnico:'Adriano Fran', horario:'10:00-21:00'});
  }
  // SBV dom — Paulo André (se disponível)
  if(trabVerde && !geTecnicoEmFerias('Paulo Andre', sDom)){
    plantoes.push({dia:'DOM', setor:'SBV', tecnico:'Paulo Andre', horario:'08:00-17:00'});
  }
  // Noturno Filial dom — Rodrigo (a cada 4 semanas)
  if(numSemana%4===0){
    if(!geTecnicoEmFerias('Rodrigo', sDom)){
      plantoes.push({dia:'DOM', setor:'NTF', tecnico:'Rodrigo', horario:'22:00-07:00'});
    }
  }

  // Avançar ciclos
  ciclo.laranja = (ciclo.laranja+1)%3;
  ciclo.azul = (ciclo.azul+1)%3;
  ciclo.verde = (ciclo.verde+1)%3;

  return plantoes;
}

function geRenderEscala(semanas, ano){
  var grid = document.getElementById('geGridEscala');
  var setores = ['SRM','RHP','ERM','SPO','SBV','TRF','NTF'];
  var coresSetor = {SRM:'#7B2FBE',RHP:'#2563eb',ERM:'#16a34a',SPO:'#E84A1A',SBV:'#d97706',TRF:'#6b7280',NTF:'#1e293b'};

  var html = '<div style="display:grid;grid-template-columns:repeat(auto-fill,minmax(320px,1fr));gap:1rem;">';

  semanas.forEach(function(sem){
    var dSab = new Date(sem.sabado);
    var dDom = new Date(sem.domingo);
    var fmtSab = dSab.getDate()+'/'+(dSab.getMonth()+1);
    var fmtDom = dDom.getDate()+'/'+(dDom.getMonth()+1);

    html += '<div style="background:#1a0f35;border:1px solid #2a1a4e;border-radius:8px;overflow:hidden;">';
    html += '<div style="background:#2a1a4e;padding:.4rem .7rem;font-size:.8rem;color:#fff;font-weight:600;">Semana '+sem.numSemana+' — '+fmtSab+' e '+fmtDom+'/'+String(ano).slice(2)+'</div>';

    // Sábado
    var sabPlantoes = sem.plantoes.filter(function(p){ return p.dia==='SAB'; });
    html += '<div style="padding:.4rem .7rem;border-bottom:1px solid #2a1a4e;">';
    html += '<div style="color:#E84A1A;font-size:.75rem;font-weight:700;margin-bottom:.3rem;">SÁBADO</div>';
    setores.forEach(function(s){
      var ps = sabPlantoes.filter(function(p){ return p.setor===s; });
      if(ps.length===0) return;
      html += '<div style="display:flex;align-items:flex-start;gap:.4rem;margin-bottom:.2rem;">';
      html += '<span style="background:'+coresSetor[s]+';color:#fff;border-radius:3px;padding:.1rem .3rem;font-size:.68rem;font-weight:700;min-width:30px;text-align:center;">'+s+'</span>';
      html += '<div>';
      ps.forEach(function(p){
        var tec = GE_TECNICOS.find(function(t){ return t.nome===p.tecnico; });
        var cor = tec ? GE_CORES[tec.grupo] : '#fff';
        html += '<div style="color:'+cor+';font-size:.75rem;">'+p.tecnico+' <span style="color:#666;">'+p.horario+'</span></div>';
      });
      html += '</div></div>';
    });
    html += '</div>';

    // Domingo
    var domPlantoes = sem.plantoes.filter(function(p){ return p.dia==='DOM'; });
    if(domPlantoes.length > 0){
      html += '<div style="padding:.4rem .7rem;">';
      html += '<div style="color:#7B2FBE;font-size:.75rem;font-weight:700;margin-bottom:.3rem;">DOMINGO</div>';
      setores.forEach(function(s){
        var ps = domPlantoes.filter(function(p){ return p.setor===s; });
        if(ps.length===0) return;
        html += '<div style="display:flex;align-items:flex-start;gap:.4rem;margin-bottom:.2rem;">';
        html += '<span style="background:'+coresSetor[s]+';color:#fff;border-radius:3px;padding:.1rem .3rem;font-size:.68rem;font-weight:700;min-width:30px;text-align:center;">'+s+'</span>';
        html += '<div>';
        ps.forEach(function(p){
          var tec = GE_TECNICOS.find(function(t){ return t.nome===p.tecnico; });
          var cor = tec ? GE_CORES[tec.grupo] : '#fff';
          html += '<div style="color:'+cor+';font-size:.75rem;">'+p.tecnico+' <span style="color:#666;">'+p.horario+'</span></div>';
        });
        html += '</div></div>';
      });
      html += '</div>';
    }
    html += '</div>';
  });

  html += '</div>';
  grid.innerHTML = html;
}
